-- Obtener el jugador más cercano
local function GetClosestPlayer()
local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
if not hrp then return nil end
local closestChar, closestDist = nil, math.huge
for i, player in pairs(Players:GetPlayers()) do
if player ~= LocalPlayer and (player.Character and player.Character:FindFirstChild("HumanoidRootPart")) then
local targetHrp = player.Character:FindFirstChildOfClass("HumanoidRootPart")
if targetHrp and targetHrp.Health > 0 then
local dist = ((player.Character.HumanoidRootPart.Position - hrp.Position).Magnitude)
if dist < closestDist and dist < MaxDistance then
closestDist = dist
closestChar = player.Character.HumanoidRootPart
end
end
end
end
return closestChar
end

if dist < closestDist and dist < MaxDistance then
closestDist = dist

end
end
return closestNpc
end

local function CreateToggle(label, y, parent)
local container = Instance.new("Frame", parent or SettingsBox)
container.Size = UDim2.new(0, 340, 0, 15)
container.Position = UDim2.new(0, 0, 0, y)
container.BackgroundTransparency = 1

title.Size = UDim2.new(0.6, 0, 0, 15)
title.BackgroundTransparency = 1
title.Text = label
title.TextColor3 = Accents.ButtonColor
title.Font = Enum.Font.Gotham
title.TextXAlignment = Enum.TextXAlignment.Left
title.TextSize = 12

local toggleBtn = Instance.new("TextButton", container)
toggleBtn.Size = UDim2.new(0, 15, 0, 15)
toggleBtn.Position = UDim2.new(1, -15, 0, 0)
toggleBtn.BackgroundColor3 = Accents.ToggleOff
toggleBtn.Text = ""
toggleBtn.AutoButtonColor = false
Instance.new("UICorner", toggleBtn).CornerRadius = UDim.new(0, 3)

local state = false
local function UpdateVisual()
if state then
toggleBtn.BackgroundColor3 = Colors.Primary
toggleBtn.Text = "ON"
if not table.find(Toggles, toggleBtn) then table.insert(Toggles, toggleBtn) end
else
toggleBtn.BackgroundColor3 = Accents.ToggleOff
toggleBtn.Text = ""
local idx = table.find(Toggles, toggleBtn)
if idx then table.remove(Toggles, idx) end
end
end

local function SetState(val)
if state == val then return end
state = val
UpdateVisual()
end

toggleBtn.MouseButton1Click:Connect(function()
state = not state
UpdateVisual()
end)

return SetState, toggleBtn
end

for i, v in ipairs(Clickables) do
if v and v:IsA("TextButton") then v.TextColor3 = Colors.Primary end

for i, v in ipairs(Toggles) do
if v and v:IsA("TextLabel") then v.TextColor3 = Colors.Primary end

end
for i, v in ipairs(ColorPickerFrame:GetChildren()) do
if v:IsA("TextButton") and Colors[v.Name] then
v.BackgroundColor3 = Colors[v.Name]
end
end
ColorPickerFrame.Visible = false

end
